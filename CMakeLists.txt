cmake_minimum_required(VERSION 3.22)
project(SDJuno6 VERSION 0.1.0)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(SDJuno6 COMPANY_NAME "Open Circuit" BUNDLE_ID com.opencircuit.juno6 PLUGIN_MANUFACTURER_CODE OpCi PLUGIN_CODE Juno FORMATS VST3 Standalone PRODUCT_NAME "SD Juno-6" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE)
target_sources(SDJuno6 PRIVATE Source/PluginProcessor.cpp)
target_compile_features(SDJuno6 PRIVATE cxx_std_17)
target_compile_definitions(SDJuno6 PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(SDJuno6 PRIVATE juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
